@njit(cache=False)
def policy_features_nb(bb,mb,st,move,undo,net,nn_last,nn_acc,before):
    f=np.zeros(32,np.float32)
    side=np.int32(st[SIDE]);sign=1 if side==0 else -1
    frm=m_from(move);to=m_to(move);pt=np.int32(mb[frm])%6
    f[pt]=1
    if m_cap(move):f[6+victim_pt(mb,move)]=1
    promo=m_promo(move)
    if promo:f[11+promo]=1
    f[17]=m_castle(move)!=0
    ph=min(24,np.int64(st[PHASE_ACC]))/24.
    base=(np.int64(st[MG_ACC])*ph+np.int64(st[EG_ACC])*(1-ph))*sign+10
    f[20]=(PAWN_A[side,to]&bb[(1-side)*6])!=0
    f[21]=(PAWN_A[1-side,to]&bb[side*6])!=0
    if pt==0:f[22]=((to>>3)-(frm>>3))*sign
    own=lsb(bb[side*6+5]);enemy=lsb(bb[(1-side)*6+5])
    f[23]=max(abs((frm&7)-(enemy&7)),abs((frm>>3)-(enemy>>3)))-max(abs((to&7)-(enemy&7)),abs((to>>3)-(enemy>>3)))
    if pt!=5:f[24]=max(abs((frm&7)-(own&7)),abs((frm>>3)-(own>>3)))-max(abs((to&7)-(own&7)),abs((to>>3)-(own>>3)))
    f[25]=ph
    f[26]=(abs(2*(frm&7)-7)+abs(2*(frm>>3)-7)-abs(2*(to&7)-7)-abs(2*(to>>3)-7))/4.
    f[27]=((frm>>3) if side==0 else 7-(frm>>3))/7.
    f[28]=((to>>3) if side==0 else 7-(to>>3))/7.
    f[29]=(frm&7)/7.;f[30]=(to&7)/7.;f[31]=m_cap(move)!=0
    make_nb(bb,mb,st,move,undo)
    f[16]=in_check_nb(bb,st,np.int32(st[SIDE]))
    newph=min(24,np.int64(st[PHASE_ACC]))/24.
    afterbase=(np.int64(st[MG_ACC])*newph+np.int64(st[EG_ACC])*(1-newph))*sign-10
    f[18]=(afterbase-base)/100.
    after=-evaluate_nb(bb,st,False,net,nn_last,nn_acc)
    f[19]=(after-before)/400.
    unmake_nb(bb,mb,st,move,undo)
    return f

@njit(cache=False)
def policy_moves_nb(bb,mb,st,moves,n,ply,hash_move,killers,histy,undo,net,nn_last,nn_acc,policy,policy_stats):
    before=evaluate_nb(bb,st,False,net,nn_last,nn_acc)
    logits=np.empty(n,np.float64);keys=np.empty(n,np.float64);cats=np.empty(n,np.int32)
    policy_stats[0]+=1;policy_stats[1]+=n
    for i in range(n):
        move=moves[i];pt=np.int32(mb[m_from(move)])%6
        f=policy_features_nb(bb,mb,st,move,undo,net,nn_last,nn_acc,before)
        v=0.
        for j in range(32):v+=policy[j]*f[j]
        for j in range(16):v+=policy[32+pt*16+j]*f[16+j]
        logits[i]=v
        cat,k=order_key(mb,move,ply,hash_move,killers,histy);cats[i]=cat
        keys[i]=k-1024*v if cat==4 else k
    # Keep TT/tactical/killer precedence; learned ranking affects every other quiet.
    for i in range(1,n):
        mv=moves[i];k=keys[i];cat=cats[i];logit=logits[i];j=i
        while j>0 and (cats[j-1]>cat or (cats[j-1]==cat and keys[j-1]>k)):
            moves[j]=moves[j-1];keys[j]=keys[j-1];cats[j]=cats[j-1];logits[j]=logits[j-1];j-=1
        moves[j]=mv;keys[j]=k;cats[j]=cat;logits[j]=logit
    maximum=np.max(logits);total=0.
    for i in range(n):logits[i]=np.exp((logits[i]-maximum)/policy[128]);total+=logits[i]
    for i in range(n):logits[i]=(1-policy[129])*logits[i]/total+policy[129]/n
    return logits

@njit(cache=False)
def learned_reduction_nb(depth,n,prob):
    # Probability controls search effort in half of the remaining nominal depth.
    # At least two full plies remain; tactical exemptions are checked by the caller.
    # Uniform probability has cost1; concentrated alternatives get more work.
    cost=max(.5,-np.log(max(prob,1e-30))/np.log(max(2,n)))
    return max(0,min(depth-3,int(np.floor(.5*(depth-1)*cost))))
